#!/usr/bin/env bash
set -Eeuo pipefail
umask 027

if [[ "${EUID}" -ne 0 ]]; then
  echo "Run this installer as root." >&2
  exit 1
fi

if [[ ! -r /etc/os-release ]]; then
  echo "Cannot identify the operating system." >&2
  exit 1
fi

source /etc/os-release
if [[ "${ID:-}" != "ubuntu" || "${VERSION_ID:-}" != "24.04" ]]; then
  echo "HYZoraX Control Center supports Ubuntu 24.04 LTS only." >&2
  exit 1
fi

bundle_dir="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
worker_mode=false

for argument in "$@"; do
  case "${argument}" in
    --worker) worker_mode=true ;;
    *)
      echo "Unknown installer option: ${argument}" >&2
      exit 2
      ;;
  esac
done

# Fresh cloud images can accept SSH while cloud-init and wait-online units are
# still running. HYZoraX does not need the global system state to be "running"
# before installation; the release has already been downloaded over HTTPS, so
# usable networking is proven. Only require a responsive systemd/D-Bus control
# plane here. Transient D-Bus resets during later systemctl operations are
# handled by bounded retries in worker mode.
wait_for_systemd_bus() {
  local waited=0
  local timeout=10
  local dbus_state=""

  if ! command -v systemctl >/dev/null 2>&1 || [[ "$(cat /proc/1/comm 2>/dev/null || true)" != "systemd" ]]; then
    return 0
  fi

  echo "Checking systemd control bus..."
  while (( waited <= timeout )); do
    dbus_state="$(systemctl is-active dbus.service 2>/dev/null || true)"
    if [[ "${dbus_state}" == "active" ]] && systemctl show --property=SystemState --value >/dev/null 2>&1; then
      echo "Systemd control bus: ready"
      return 0
    fi
    sleep 1
    waited=$((waited + 1))
  done

  echo "Systemd control bus did not become ready within ${timeout}s (dbus=${dbus_state:-unknown})." >&2
  return 1
}

transient_unit_known() {
  local unit_name="$1"
  local load_state=""
  load_state="$(systemctl show --property=LoadState --value "${unit_name}.service" 2>/dev/null || true)"
  [[ -n "${load_state}" && "${load_state}" != "not-found" ]]
}

launch_transient_worker() {
  local unit_name="$1"
  local status_file="$2"
  shift 2
  local attempt=1
  local max_attempts=5
  local output=""

  while (( attempt <= max_attempts )); do
    if transient_unit_known "${unit_name}"; then
      return 0
    fi

    if output="$(systemd-run --quiet --no-block \
      --unit="${unit_name}" \
      --property=Type=oneshot \
      --setenv="HYZORAX_INSTALL_STATUS_FILE=${status_file}" \
      /usr/bin/bash "${bundle_dir}/bootstrap.sh" "$@" 2>&1)"; then
      [[ -z "${output}" ]] || printf '%s\n' "${output}"
      return 0
    fi

    printf '%s\n' "${output}" >&2
    sleep 1
    if transient_unit_known "${unit_name}"; then
      echo "Transient installer was accepted despite a control-bus interruption; continuing." >&2
      return 0
    fi

    if grep -Eqi 'Connection reset by peer|D-Bus connection terminated|Failed to connect to bus|Transport endpoint is not connected' <<<"${output}" && (( attempt < max_attempts )); then
      echo "Transient systemd launch hit a temporary bus failure; retrying (${attempt}/${max_attempts})..." >&2
      sleep 1
      attempt=$((attempt + 1))
      continue
    fi
    return 1
  done
}
# Launch the mutating work as a transient systemd unit. If SSH disappears, the
# installation continues and writes a root-only recovery log.
if [[ "${worker_mode}" == false ]] && command -v systemd-run >/dev/null 2>&1 && [[ "$(cat /proc/1/comm 2>/dev/null || true)" == "systemd" ]]; then
  wait_for_systemd_bus
  unit_name="hyzorax-control-install-$(date -u +%Y%m%d%H%M%S)-$$"
  status_file="/run/${unit_name}.status"
  worker_arguments=(--worker)
  launch_transient_worker "${unit_name}" "${status_file}" "${worker_arguments[@]}"

  echo "HYZoraX installation started as ${unit_name}.service"
  echo "It will continue safely if this SSH session disconnects."
  echo "Waiting for completion..."
  wait_seconds=0
  while [[ ! -f "${status_file}" ]]; do
    if systemctl is-failed --quiet "${unit_name}.service"; then
      echo "Installation worker failed before returning its status." >&2
      journalctl --no-pager -u "${unit_name}.service" >&2 || true
      exit 1
    fi
    if (( wait_seconds >= 900 )); then
      echo "Timed out waiting for the installation worker. It may still be inspected through systemd." >&2
      exit 1
    fi
    sleep 1
    wait_seconds=$((wait_seconds + 1))
  done
  install_status="$(<"${status_file}")"
  if [[ -r /var/log/hyzorax-control-install.log ]]; then
    echo
    cat /var/log/hyzorax-control-install.log
  fi
  if [[ "${install_status}" != "0" ]]; then
    echo "HYZoraX installation failed with status ${install_status}." >&2
    exit "${install_status}"
  fi
  exit 0
fi

install_log="/var/log/hyzorax-control-install.log"
credentials_cache="/etc/hyzorax-control/owner.credentials"
legacy_owner_username=""
legacy_owner_password=""
if [[ ! -r "${credentials_cache}" && -r "${install_log}" ]]; then
  legacy_owner_username="$(grep -E '^Initial Owner username:' "${install_log}" 2>/dev/null | tail -n 1 | sed 's/^Initial Owner username:[[:space:]]*//' || true)"
  legacy_owner_password="$(grep -E '^Initial Owner password:' "${install_log}" 2>/dev/null | tail -n 1 | sed 's/^Initial Owner password:[[:space:]]*//' || true)"
fi
install -o root -g root -m 0600 /dev/null "${install_log}"
exec > >(/usr/bin/tee "${install_log}") 2>&1

finish_install() {
  status=$?
  trap - EXIT
  if [[ -n "${HYZORAX_INSTALL_STATUS_FILE:-}" ]]; then
    printf '%s\n' "${status}" > "${HYZORAX_INSTALL_STATUS_FILE}"
    chmod 0600 "${HYZORAX_INSTALL_STATUS_FILE}"

  exit "${status}"
}

generate_entrance_code() {
  local chars="abcdefghjkmnpqrstuvwxyz23456789"
  local code=""
  local value=0
  local index=0
  while true; do
    code=""
    while (( ${#code} < 8 )); do
      value="$(od -An -N1 -tu1 /dev/urandom | tr -d ' ')"
      [[ "${value}" =~ ^[0-9]+$ ]] || continue
      index=$((value % ${#chars}))
      code+="${chars:index:1}"
    done
    if [[ "${code}" =~ [a-z] && "${code}" =~ [0-9] ]]; then
      printf '%s' "${code}"
      return 0
    fi
  done
}

valid_entrance_code() {
  local code="${1:-}"
  [[ "${code}" =~ ^[a-z0-9]{8}$ && "${code}" =~ [a-z] && "${code}" =~ [0-9] ]]
}

systemctl_retry() {
  local attempt=1
  local max_attempts=5
  local output=""
  while (( attempt <= max_attempts )); do
    if output="$(systemctl "$@" 2>&1)"; then
      [[ -z "${output}" ]] || printf '%s\n' "${output}"
      return 0
    fi

    printf '%s\n' "${output}" >&2
    if grep -Eqi 'Connection reset by peer|D-Bus connection terminated|Failed to connect to bus|Transport endpoint is not connected' <<<"${output}" && (( attempt < max_attempts )); then
      echo "systemctl transient bus failure; retrying (${attempt}/${max_attempts})..." >&2
      sleep 2
      attempt=$((attempt + 1))
      continue
    fi
    return 1
  done
}

#!/usr/bin/env bash
set -Eeuo pipefail

entrance_code_file="/etc/hyzorax-control/entrance.code"
credentials_cache="/etc/hyzorax-control/owner.credentials"
tls_manager="/usr/local/libexec/hyzorax-control-tls"
updater="/usr/local/libexec/hyzorax-control-updater"

if [[ "${EUID}" -ne 0 ]]; then
  echo "Run hz as root." >&2
  exit 1
fi

panel_host() {
  local detected=""
  if command -v ip >/dev/null 2>&1; then
    detected="$(ip -4 route get 1.1.1.1 2>/dev/null | awk '{for (i=1; i<=NF; i++) if ($i == "src") {print $(i+1); exit}}' || true)"
  fi
  [[ -n "${detected}" ]] || detected="$(hostname -I 2>/dev/null | awk '{print $1}' || true)"
  [[ -n "${detected}" ]] || detected="$(hostname -f 2>/dev/null || hostname)"
  [[ "${detected}" == *:* ]] && detected="[${detected}]"
  printf '%s' "${detected}"
}

valid_entrance_code() {
  local code="${1:-}"
  [[ "${code}" =~ ^[a-z0-9]{8}$ && "${code}" =~ [a-z] && "${code}" =~ [0-9] ]]
}

current_entrance_code() {
  local code=""
  [[ -r "${entrance_code_file}" ]] || { echo "Security entrance code is not available." >&2; return 1; }
  code="$(<"${entrance_code_file}")"
  valid_entrance_code "${code}" || { echo "Stored security entrance code is invalid." >&2; return 1; }
  printf '%s' "${code}"
}

show_url() {
  local entrance_code=""
  entrance_code="$(current_entrance_code)" || return 1
  printf 'Panel URL: https://%s:9443/%s/\n' "$(panel_host)" "${entrance_code}"
}

read_credentials_cache() {
  cached_username=""
  cached_password=""
  [[ -r "${credentials_cache}" ]] || return 1
  cached_username="$(sed -n 's/^username=//p' "${credentials_cache}" | head -n 1)"
  cached_password="$(sed -n 's/^password=//p' "${credentials_cache}" | head -n 1)"
  [[ -n "${cached_username}" && -n "${cached_password}" ]]
}

write_credentials_cache() {

generate_entrance_code() {
  local chars="abcdefghjkmnpqrstuvwxyz23456789"
  local code="" value=0 index=0
  while true; do
    code=""
    while (( ${#code} < 8 )); do
      value="$(od -An -N1 -tu1 /dev/urandom | tr -d ' ')"
      [[ "${value}" =~ ^[0-9]+$ ]] || continue
      index=$((value % ${#chars}))
      code+="${chars:index:1}"
    done
    [[ "${code}" =~ [a-z] && "${code}" =~ [0-9] ]] && { printf '%s' "${code}"; return 0; }
  done
}

    curl -kfsS --max-time 2 "https://127.0.0.1:9443/${code}/api/v1/healthz" >/dev/null 2>&1 && return 0
    sleep 1
  done
  return 1
}

restore_entrance_code() {
  local code="${1:-}" tmp=""
  tmp="$(mktemp /etc/hyzorax-control/.entrance-code.XXXXXX)"
  printf '%s\n' "${code}" > "${tmp}"
  chown root:hyzorax-control "${tmp}"
  chmod 0640 "${tmp}"
  mv -f "${tmp}" "${entrance_code_file}"
}

apply_entrance_code() {
  local new_code="${1:-}" old_code="" tmp=""
  valid_entrance_code "${new_code}" || { echo "Security entrance must be exactly 8 lowercase letters/numbers and include at least one letter and one number." >&2; return 1; }
  old_code="$(current_entrance_code)" || return 1
  if [[ "${new_code}" == "${old_code}" ]]; then
    echo "Security entrance is already ${new_code}."
    show_url
    return 0
  fi
  tmp="$(mktemp /etc/hyzorax-control/.entrance-code.XXXXXX)"
  printf '%s\n' "${new_code}" > "${tmp}"
  chown root:hyzorax-control "${tmp}"
  chmod 0640 "${tmp}"
  mv -f "${tmp}" "${entrance_code_file}"
  if ! systemctl restart hyzorax-control.service || ! wait_for_entrance_health "${new_code}"; then
    restore_entrance_code "${old_code}"
    systemctl restart hyzorax-control.service || true
    echo "Security entrance change failed and the previous code was restored." >&2
    return 1
  fi
  echo "Security entrance updated. The previous panel URL is no longer valid."
  show_url
}

detect_panel_ip() {
  local detected=""
  if command -v ip >/dev/null 2>&1; then
    detected="$(ip -4 route get 1.1.1.1 2>/dev/null | awk '{for (i=1; i<=NF; i++) if ($i == "src") {print $(i+1); exit}}' || true)"
  fi
  if [[ -z "${detected}" ]]; then
    detected="$(hostname -I 2>/dev/null | awk '{for (i=1; i<=NF; i++) if ($i ~ /^[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+$/) {print $i; exit}}' || true)"
  fi
  if [[ -z "${detected}" ]]; then
    fail "Could not detect an IPv4 address for the panel." || return 1
  fi
  if ! python3 - "${detected}" <<'PY'
import ipaddress, sys
try:
    ip = ipaddress.ip_address(sys.argv[1])
except ValueError:
    raise SystemExit(1)
if ip.version != 4 or not ip.is_global:
    raise SystemExit(1)
PY
  then
    fail "Detected address ${detected} is not a globally routable IPv4 address." || return 1
  fi
  printf '%s' "${detected}"
}

entrance_code() {
  local code=""
  [[ -r "${entrance_code_file}" ]] || return 1
  code="$(<"${entrance_code_file}")"
  [[ "${code}" =~ ^[a-z0-9]{8}$ && "${code}" =~ [a-z] && "${code}" =~ [0-9] ]] || return 1
  printf '%s' "${code}"
}

      return 0
    fi
    sleep 1
  done
  return 1
}

port_80_is_free() {
  if command -v ss >/dev/null 2>&1; then
    [[ -z "$(ss -H -ltn 'sport = :80' 2>/dev/null || true)" ]]
    return
  fi
  python3 - <<'PY'
import socket
s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
try:
    s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    s.bind(("0.0.0.0", 80))
except OSError:
    raise SystemExit(1)
finally:
    s.close()
PY
}

show_port_80_owner() {

apt_lock_error() {
  grep -Eqi 'Could not get lock|Unable to acquire.*lock|is another process using it|Could not open lock file|Resource temporarily unavailable' <<<"${1:-}"
}

run_apt() {
  local mode="${1:-quick}"
  shift
  local output=""
  local rc=0
  local waited=0
  local timeout=1200

  while true; do
    if output="$(DEBIAN_FRONTEND=noninteractive apt-get "$@" 2>&1)"; then
      [[ -z "${output}" ]] || printf '%s
' "${output}"
      return 0
    else
      rc=$?
    fi

    if ! apt_lock_error "${output}"; then
      printf '%s
' "${output}" >&2
      return "${rc}"
    fi

    if [[ "${mode}" != "wait" ]]; then
      warn "Ubuntu package manager is busy; trusted IP TLS will be retried automatically in the background."
      return 75
    fi

    if (( waited >= timeout )); then
      warn "Ubuntu package-manager lock stayed busy for ${timeout}s; leaving current TLS untouched."
      return 75
    fi

    if (( waited == 0 || waited % 30 == 0 )); then
      log "Ubuntu package manager is busy; waiting in the background before preparing trusted IP TLS..."
    fi
    sleep 5
    waited=$((waited + 5))
  done
}

ensure_certbot() {
  local mode="${HYZORAX_TLS_CERTBOT_MODE:-quick}"
  local rc=0

  if [[ -x "${certbot_bin}" ]] && "${certbot_bin}" --version 2>/dev/null | grep -q "${certbot_version}"; then
    return 0
  fi

  log "Preparing isolated Certbot ${certbot_version} client under ${client_root}."

  # A failed `python3 -m venv` can leave a partial directory. Never reuse it.
  rm -rf -- "${client_root}"
  if ! python3 -m venv "${client_root}" >/dev/null 2>&1; then
    rm -rf -- "${client_root}"
    if ! command -v apt-get >/dev/null 2>&1; then
      fail "Python venv support is unavailable and apt-get is not present." || return 1
    fi

    log "Preparing Ubuntu python3-venv dependency."
    run_apt "${mode}" update || { rc=$?; return "${rc}"; }
    run_apt "${mode}" install -y python3-venv || { rc=$?; return "${rc}"; }

    rm -rf -- "${client_root}"
    if ! python3 -m venv "${client_root}" >/dev/null 2>&1; then
      rm -rf -- "${client_root}"
      fail "python3-venv is installed but the isolated TLS virtual environment still could not be created." || return 1
    fi
  fi

  if [[ ! -x "${client_root}/bin/python" ]] || ! "${client_root}/bin/python" -m pip --version >/dev/null 2>&1; then
    rm -rf -- "${client_root}"
    fail "The isolated TLS Python environment is incomplete." || return 1
  fi

  if ! "${client_root}/bin/python" -m pip install --disable-pip-version-check --no-cache-dir --upgrade "pip<26" >/dev/null; then
    fail "Could not prepare pip inside the isolated TLS client." || return 1
  fi
  if ! "${client_root}/bin/pip" install --disable-pip-version-check --no-cache-dir "certbot==${certbot_version}"; then
    fail "Could not install the pinned Certbot ${certbot_version} client." || return 1
  fi
  "${certbot_bin}" --version >/dev/null 2>&1 || { fail "The isolated Certbot client failed its version check."; return 1; }
  log "Isolated Certbot ${certbot_version} client is ready."
}

managed_live_dir() {
  printf '%s/live/%s' "${acme_config}" "${cert_name}"
}

validate_pair() {
  local cert_file="${1:?certificate path required}"
  local key_file="${2:?key path required}"
  local ip_addr="${3:?IP address required}"
  [[ -r "${cert_file}" && -r "${key_file}" ]] || return 1
  openssl x509 -in "${cert_file}" -noout -checkip "${ip_addr}" >/dev/null 2>&1 || return 1
  openssl x509 -in "${cert_file}" -noout -checkend 21600 >/dev/null 2>&1 || return 1
  local cert_pub key_pub
  cert_pub="$(openssl x509 -in "${cert_file}" -pubkey -noout 2>/dev/null | openssl pkey -pubin -outform DER 2>/dev/null | sha256sum | awk '{print $1}')"
  key_pub="$(openssl pkey -in "${key_file}" -pubout -outform DER 2>/dev/null | sha256sum | awk '{print $1}')"
  [[ -n "${cert_pub}" && "${cert_pub}" == "${key_pub}" ]]
}

deploy_managed_certificate() {
  local ip_addr="${1:?IP address required}"
  local live_dir cert_source key_source cert_hash current_hash
  live_dir="$(managed_live_dir)"
  cert_source="${live_dir}/fullchain.pem"
  key_source="${live_dir}/privkey.pem"
  validate_pair "${cert_source}" "${key_source}" "${ip_addr}" || {
    fail "The newly issued certificate/key pair failed local IP and key validation." || return 1
  }

  cert_hash="$(sha256sum "${cert_source}" | awk '{print $1}')"
  current_hash=""
  if [[ -r "${panel_cert}" ]]; then
    current_hash="$(sha256sum "${panel_cert}" | awk '{print $1}')"
  fi
  if [[ "${cert_hash}" == "${current_hash}" ]] && validate_pair "${panel_cert}" "${panel_key}" "${ip_addr}"; then
    touch "${managed_marker}"
    return 0
  fi

  local stamp backup_dir cert_tmp key_tmp had_old=false
  stamp="$(date -u +%Y%m%dT%H%M%SZ)"
  backup_dir="${backup_root}/${stamp}"
  install -d -o root -g root -m 0700 "${backup_dir}"
  if [[ -r "${panel_cert}" && -r "${panel_key}" ]]; then
    cp -a -- "${panel_cert}" "${backup_dir}/panel.crt"
    cp -a -- "${panel_key}" "${backup_dir}/panel.key"
    had_old=true
  fi

  cert_tmp="$(mktemp "${panel_tls_dir}/.panel.crt.XXXXXX")"
  key_tmp="$(mktemp "${panel_tls_dir}/.panel.key.XXXXXX")"
  install -o root -g hyzorax-control -m 0644 "${cert_source}" "${cert_tmp}"
  install -o root -g hyzorax-control -m 0640 "${key_source}" "${key_tmp}"
  mv -f -- "${cert_tmp}" "${panel_cert}"
  mv -f -- "${key_tmp}" "${panel_key}"

  if systemctl restart "${panel_service}" && health_check; then
    touch "${managed_marker}"
    chmod 0600 "${managed_marker}"
    log "Trusted IP certificate deployed for ${ip_addr}; Control Center health check passed."
    return 0
  fi

  warn "Control Center failed after TLS deployment; restoring the previous certificate."
  if [[ "${had_old}" == true ]]; then
    install -o root -g hyzorax-control -m 0644 "${backup_dir}/panel.crt" "${panel_cert}"
    install -o root -g hyzorax-control -m 0640 "${backup_dir}/panel.key" "${panel_key}"
  fi
  systemctl restart "${panel_service}" || true
  health_check || true
  return 1
}

issue_new_certificate() {
  local ip_addr="${1:?IP address required}"
  port_80_is_free || {
    warn "Public IP validation needs TCP port 80, but a local listener is already using it."
    show_port_80_owner
    warn "HYZoraX will not stop or reconfigure Nginx, Apache, any website, domain, subdomain, or mail service automatically."
    return 2
  }
  ensure_certbot || return 1
  log "Requesting a Let's Encrypt short-lived certificate for IP ${ip_addr}."
  "${certbot_bin}" certonly \
    --standalone \
    --non-interactive \
    --agree-tos \
    --register-unsafely-without-email \
    --required-profile shortlived \
    --preferred-challenges http \
    --ip-address "${ip_addr}" \
    --cert-name "${cert_name}" \
    --config-dir "${acme_config}" \
    --work-dir "${acme_work}" \
    --logs-dir "${acme_logs}" \
    --key-type ecdsa \
    --elliptic-curve secp256r1
  deploy_managed_certificate "${ip_addr}"
}

renew_certificate() {
  local ip_addr="${1:?IP address required}"
  local live_dir
  live_dir="$(managed_live_dir)"
  if [[ ! -r "${live_dir}/fullchain.pem" || ! -r "${live_dir}/privkey.pem" ]]; then
    issue_new_certificate "${ip_addr}"
    return $?
  fi
  port_80_is_free || {
    warn "Renewal is due through HTTP-01 when Certbot decides it is near expiry, but TCP port 80 is currently occupied."
    show_port_80_owner
    return 2
  }
  ensure_certbot || return 1
  "${certbot_bin}" renew \
    --cert-name "${cert_name}" \
    --config-dir "${acme_config}" \
    --work-dir "${acme_work}" \
    --logs-dir "${acme_logs}" \
    --no-directory-hooks
  deploy_managed_certificate "${ip_addr}"
}

make_self_signed_fallback() {
  local ip_addr="${1:?IP address required}"
  local tmpdir backup_dir stamp
  tmpdir="$(mktemp -d)"
  openssl req -x509 -newkey rsa:3072 -sha256 -nodes -days 30 \
    -subj "/CN=${ip_addr}" \
    -addext "subjectAltName=IP:${ip_addr}" \
    -keyout "${tmpdir}/panel.key" \
    -out "${tmpdir}/panel.crt" >/dev/null 2>&1
  validate_pair "${tmpdir}/panel.crt" "${tmpdir}/panel.key" "${ip_addr}" || return 1
  stamp="$(date -u +%Y%m%dT%H%M%SZ)"
  backup_dir="${backup_root}/${stamp}-before-self-signed"
  install -d -o root -g root -m 0700 "${backup_dir}"
  [[ -r "${panel_cert}" ]] && cp -a -- "${panel_cert}" "${backup_dir}/panel.crt" || true
  [[ -r "${panel_key}" ]] && cp -a -- "${panel_key}" "${backup_dir}/panel.key" || true
  install -o root -g hyzorax-control -m 0644 "${tmpdir}/panel.crt" "${panel_cert}"
  install -o root -g hyzorax-control -m 0640 "${tmpdir}/panel.key" "${panel_key}"
  rm -f -- "${managed_marker}"
  systemctl restart "${panel_service}"
  health_check
  rm -rf -- "${tmpdir}"
  log "Self-signed IP fallback certificate installed. Browser trust warnings are expected."
}

status() {
  local ip_addr="unknown"
  ip_addr="$(detect_panel_ip 2>/dev/null || printf 'unknown')"
  echo "HYZoraX Control Center IP TLS"
  echo "Panel IP: ${ip_addr}"
  echo "Panel certificate: ${panel_cert}"
  if [[ -r "${panel_cert}" ]]; then
    openssl x509 -in "${panel_cert}" -noout -subject -issuer -dates 2>/dev/null || true
    openssl x509 -in "${panel_cert}" -noout -ext subjectAltName 2>/dev/null || true
    if [[ "${ip_addr}" != "unknown" ]] && openssl x509 -in "${panel_cert}" -noout -checkip "${ip_addr}" >/dev/null 2>&1; then
      echo "IP SAN match: yes"
    else
      echo "IP SAN match: no"
    fi
  else
    echo "Certificate file is missing."
  fi
  if [[ -f "${managed_marker}" ]]; then
    echo "Trusted-IP management: enabled"
  else
    echo "Trusted-IP management: not yet active"
  fi
  if port_80_is_free; then
    echo "TCP port 80: free for independent HTTP-01 validation"
  else
    echo "TCP port 80: occupied; HYZoraX will not stop that service automatically"
    show_port_80_owner
  fi
  if systemctl is-enabled --quiet hyzorax-control-tls-renew.timer 2>/dev/null; then
    echo "Automatic renewal timer: enabled"
    systemctl --no-pager list-timers hyzorax-control-tls-renew.timer 2>/dev/null | tail -n +2 || true
  else
    echo "Automatic renewal timer: disabled"
  fi
}

maintain() {
  prepare_directories
  local ip_addr
  ip_addr="$(detect_panel_ip)" || return 1
  if [[ -r "$(managed_live_dir)/fullchain.pem" ]]; then
    renew_certificate "${ip_addr}"
  else
    issue_new_certificate "${ip_addr}"
  fi
}

bootstrap_try() {
  prepare_directories
  local ip_addr rc=0
  ip_addr="$(detect_panel_ip)" || { warn "Trusted IP TLS was not attempted because no public IPv4 address was detected."; return 1; }

  HYZORAX_TLS_CERTBOT_MODE=quick ensure_certbot || {
    rc=$?
    if (( rc == 75 )); then
      return 75
    fi
    warn "The isolated Certbot client could not be prepared. Existing TLS remains active."
    return "${rc}"
  }

  if ! port_80_is_free; then
    warn "Trusted IP TLS cannot be issued while TCP port 80 is occupied. Existing TLS remains untouched."
    return 2
  fi

  if ! maintain; then
    warn "Trusted IP certificate provisioning did not complete. Existing TLS remains active."
    return 1
  fi
}

bootstrap_wait() {
  prepare_directories
  local ip_addr rc=0
  ip_addr="$(detect_panel_ip)" || { warn "Background trusted-IP TLS bootstrap could not detect a public IPv4 address."; return 1; }

  if ! port_80_is_free; then
    warn "Background trusted-IP TLS bootstrap found TCP port 80 occupied. Existing TLS remains active; 'hz ssl' can retry later."
    show_port_80_owner
    return 2
  fi

  HYZORAX_TLS_CERTBOT_MODE=wait ensure_certbot || {
    rc=$?
    warn "Background trusted-IP TLS bootstrap could not prepare Certbot. Existing TLS remains active; 'hz ssl' can retry later."
    return "${rc}"
  }

  if [[ -r "$(managed_live_dir)/fullchain.pem" ]]; then
    renew_certificate "${ip_addr}"
  else
    issue_new_certificate "${ip_addr}"
  fi
}

require_root
case "${1:-status}" in
  status) prepare_directories; status ;;
  enable|maintain) maintain ;;
  renew) prepare_directories; renew_certificate "$(detect_panel_ip)" ;;
  bootstrap-try) bootstrap_try ;;
  bootstrap-wait) bootstrap_wait ;;
  self-signed) prepare_directories; make_self_signed_fallback "$(detect_panel_ip)" ;;
  *)
    echo "Usage: hyzorax-control-tls [status|enable|renew|maintain|bootstrap-try|bootstrap-wait|self-signed]" >&2
    exit 2
    ;;
esac

#!/usr/bin/env bash
set -Eeuo pipefail
umask 077

PRODUCT="HYZoraX Control Center"
REPOSITORY="hyzorax/hyzorax-control-center"
RELEASE_BASE="https://github.com/${REPOSITORY}/releases/latest/download"
ARCHIVE_NAME="HYZoraX_Control_Center_Linux_x86_64.tar.gz"
CHECKSUM_NAME="${ARCHIVE_NAME}.sha256"
UPDATE_LOG="/var/log/hyzorax-control-update.log"
LOCK_FILE="/run/hyzorax-control/update.lock"

fail() {
  printf '[HYZoraX Update] ERROR: %s\n' "$*" >&2
  exit 1
}

[[ "${EUID}" -eq 0 ]] || fail "Run this updater as root."
for command_name in curl sha256sum tar awk install mktemp flock sort; do
  command -v "${command_name}" >/dev/null 2>&1 || fail "Required command is missing: ${command_name}"
done

install -d -o root -g hyzorax-control -m 0750 /run/hyzorax-control
exec 9>"${LOCK_FILE}"
if ! flock -n 9; then
  fail "Another Control Center update is already running."
fi

install -o root -g root -m 0600 /dev/null "${UPDATE_LOG}"
exec > >(/usr/bin/tee "${UPDATE_LOG}") 2>&1

echo "[HYZoraX Update] Checking the latest verified release..."
current_version="$(/usr/local/bin/hyzorax-control -version | awk '{print $NF}')"
workdir="$(mktemp -d /root/.hyzorax-self-update.XXXXXX)"
trap 'rm -rf -- "${workdir}"' EXIT
archive_path="${workdir}/${ARCHIVE_NAME}"
checksum_path="${workdir}/${CHECKSUM_NAME}"
extract_dir="${workdir}/extract"
mkdir -p "${extract_dir}"

curl_secure() {
  curl --fail --silent --show-error --location --retry 3 --retry-delay 2 \
    --connect-timeout 15 --max-time 600 --proto '=https' --tlsv1.2 "$@"
}

curl_secure --output "${checksum_path}" "${RELEASE_BASE}/${CHECKSUM_NAME}"
curl_secure --output "${archive_path}" "${RELEASE_BASE}/${ARCHIVE_NAME}"
expected_sha="$(awk 'NF {print $1; exit}' "${checksum_path}")"
[[ "${expected_sha}" =~ ^[0-9a-fA-F]{64}$ ]] || fail "Latest release checksum file is invalid."
actual_sha="$(sha256sum "${archive_path}" | awk '{print $1}')"
[[ "${actual_sha,,}" == "${expected_sha,,}" ]] || fail "Latest release archive failed SHA-256 verification."
echo "[HYZoraX Update] Outer SHA-256: OK"

tar -xzf "${archive_path}" -C "${extract_dir}"
build_dir="${extract_dir}/hyzorax-control/build"
[[ -f "${build_dir}/SHA256SUMS" ]] || fail "Latest release is missing SHA256SUMS."
(
  cd "${build_dir}"
  sha256sum --check --strict SHA256SUMS
)

candidate_version="$("${build_dir}/hyzorax-control" -version | awk '{print $NF}')"
helper_version="$("${build_dir}/hyzorax-control-helper" -version | awk '{print $NF}')"
[[ "${candidate_version}" =~ ^[0-9]+\.[0-9]+\.[0-9]+([.-][0-9A-Za-z.-]+)?$ ]] || fail "Latest Control binary returned an invalid version."
[[ "${candidate_version}" == "${helper_version}" ]] || fail "Latest Control/helper version mismatch."

if [[ "${candidate_version}" == "${current_version}" ]]; then
  echo "[HYZoraX Update] Version ${current_version} is already current."
  exit 0
fi

newest="$(printf '%s\n%s\n' "${current_version}" "${candidate_version}" | sort -V | tail -n 1)"
[[ "${newest}" == "${candidate_version}" ]] || fail "Refusing to downgrade from ${current_version} to ${candidate_version}."

release_dir="/root/hyzorax-releases/v${candidate_version}"
if [[ -e "${release_dir}" ]]; then
  existing_build="${release_dir}/hyzorax-control/build"
  [[ -f "${existing_build}/SHA256SUMS" ]] || fail "Existing staged release is incomplete: ${release_dir}"
  (cd "${existing_build}" && sha256sum --check --strict SHA256SUMS >/dev/null) || fail "Existing staged release failed integrity verification."
  build_dir="${existing_build}"
  echo "[HYZoraX Update] Reusing verified staged release v${candidate_version}."
else
  install -d -o root -g root -m 0700 "${release_dir}"
  cp -a "${extract_dir}/hyzorax-control" "${release_dir}/hyzorax-control"
  build_dir="${release_dir}/hyzorax-control/build"
  echo "[HYZoraX Update] Staged v${candidate_version}."
fi

echo "[HYZoraX Update] Updating ${PRODUCT} ${current_version} -> ${candidate_version}..."
cd "${build_dir}"
./bootstrap.sh

echo "[HYZoraX Update] Update completed successfully."
